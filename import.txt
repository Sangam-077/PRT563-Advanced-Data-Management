// AUDIT BY RELATIONSHIP TYPE
MATCH ()-[r]->() RETURN type(r) AS Relationship, count(*) AS Edges ORDER BY Relationship;


// AUDIT INDEXES AND CONSTRAINTS
SHOW INDEXES YIELD name, type, entityType, labelsOrTypes, properties RETURN name,type,entityType,labelsOrTypes,properties ORDER BY name;
